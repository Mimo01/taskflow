  export async function searchJira(baseUrl, token, projectKey, query): Promise<JiraIssue[]> {
    const base = baseUrl.replace(/\/$/, '');
    const jql = `project = ${projectKey} AND text ~ "${query}" ORDER BY updated DESC`;
    const url = `${base}/rest/api/2/search?jql=${encodeURIComponent(jql)}&fields=summary,status,...&maxResults=20`;
    let response: Response;
    try {
      response = await apiFetch('jira', url, { headers: { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' } }, 'Search Issues');
    } catch { return []; }
    if (!response.ok) { return []; }
    const data = await response.json();
    return (data.issues ?? []) as JiraIssue[];
  }
