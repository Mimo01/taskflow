**1. Create widget wrappers** that handle their own token loading (per Research pitfall 4):

Each wrapper follows this pattern -- load tokens internally from Stronghold and auth store, render Skeleton while loading, then render the underlying panel.

**`taskflow/src/routes/dashboard/widgets/SubtasksWidget.tsx`:**
```typescript
import { useEffect, useState } from 'react';
import { useOutletContext } from 'react-router-dom';
import { readSecret } from '@/services/stronghold';
import { useAuthStore } from '@/stores/auth.store';
import { Skeleton } from '@/components/ui/skeleton';
import SubtasksPanel from '../SubtasksPanel';

export default function SubtasksWidget({ widgetId }: { widgetId: string }) {
  const { jiraBaseUrl, activeJiraProject } = useAuthStore();
  const [jiraToken, setJiraToken] = useState<string | null>(null);
  // Try to get onIssueClick from outlet context (may not be available in all contexts)
  let onIssueClick: ((key: string) => void) | undefined;
  try {
    const ctx = useOutletContext<{ onIssueClick?: (key: string) => void }>();
    onIssueClick = ctx?.onIssueClick;
  } catch { /* outlet context not available */ }

  useEffect(() => {
    if (jiraBaseUrl) readSecret('jira-pat').then(setJiraToken).catch(() => setJiraToken(null));
  }, [jiraBaseUrl]);

  if (!jiraToken || !jiraBaseUrl || !activeJiraProject)
    return <div className="space-y-2 p-2"><Skeleton className="h-4 w-full" /><Skeleton className="h-4 w-3/4" /></div>;

  return <SubtasksPanel jiraBaseUrl={jiraBaseUrl} jiraToken={jiraToken} activeJiraProject={activeJiraProject} onIssueClick={onIssueClick} />;
}
```
