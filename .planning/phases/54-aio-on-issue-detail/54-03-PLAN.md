function pickLatestActiveCycle(cycles: AioCycle[]): AioCycle | undefined {
  const cycleNum = (key: string) => { const m = key.match(/CY-(\d+)$/); return m ? parseInt(m[1], 10) : -1; };
  return cycles.filter((c) => c.status === 'Active').sort((a, b) => cycleNum(b.key) - cycleNum(a.key)).at(0);

**NOTE on D-04 empty state distinction**: The queryFn returns `[]` in two cases — no test cases (section hidden) and no matched runs (EmptyState).
