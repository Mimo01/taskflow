   ```typescript
   function transformPanelListItems(panelBody: string): string {
     // Find runs of consecutive lines starting with `# ` (Jira numbered-list marker).
     // Each run becomes a single <ol>…</ol>. Lines outside runs pass through.
     const lines = panelBody.split('\n');
     const out: string[] = [];
     let i = 0;
     while (i < lines.length) {
       const line = lines[i];
       if (/^\s*#\s+/.test(line)) {
         const items: string[] = [];
         while (i < lines.length && /^\s*#\s+/.test(lines[i])) {
           items.push(lines[i].replace(/^\s*#\s+/, ''));
           i++;
         }
         // Transform each item: `[name|url]` → <a href="url">name</a>; otherwise leave as text.
         // The wiki-link transform here is necessary because once we wrap in <ol><li>,
         // jira2md will NOT see the `[name|url]` syntax (it's inside raw HTML which
         // passes through rehype-raw rather than jira2md). So we must produce the
         // anchor ourselves so markdownComponents.a picks it up.
         const liItems = items
           .map((item) => {
             const linkMatch = /^\[([^|\]]+)\|([^\]]+)\]$/.exec(item.trim());
             if (linkMatch) {
               return `<li><a href="${linkMatch[2]}">${linkMatch[1]}</a></li>`;
             }
             // Non-link item — still escape any remaining `|` to prevent table split.
             return `<li>${item.replace(/\|/g, '\\|')}</li>`;
           })
           .join('');
   ```
