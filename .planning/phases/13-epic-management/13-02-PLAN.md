function epicColorClass(epicKey: string): string {
  let hash = 0
  for (let i = 0; i < epicKey.length; i++) hash = (hash * 31 + epicKey.charCodeAt(i)) >>> 0
  const colors = ['bg-violet-100 text-violet-800 border-violet-300', 'bg-blue-100 text-blue-800 border-blue-300', /* etc */]
  return colors[hash % colors.length]
}
