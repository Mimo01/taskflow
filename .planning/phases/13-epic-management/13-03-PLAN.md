**Component structure:**
```typescript
import * as Dialog from '@base-ui/react/dialog'
import { useState } from 'react'
import { useMutation, useQueryClient } from '@tanstack/react-query'
import { createIssue } from '@/services/jira'
import { useSettingsStore } from '@/stores/settings.store'
import { readSecret } from '@/lib/stronghold'  // adjust import to match project pattern

export interface CreateEpicDialogProps {
  open: boolean
  onClose: () => void
}

export function CreateEpicDialog({ open, onClose }: CreateEpicDialogProps) {
  const { jiraBaseUrl, activeJiraProject, epicNameFieldKey } = useSettingsStore()
  const [token, setToken] = useState<string | null>(null)
  useEffect(() => { readSecret('jira-pat').then(setToken) }, [])

  const [epicName, setEpicName] = useState('')
  const [description, setDescription] = useState('')
  // priority and assignee: optional fields, omit for MVP simplicity per CONTEXT.md locked fields

  const queryClient = useQueryClient()
  const mutation = useMutation({
    mutationFn: async () => {
      if (!jiraBaseUrl || !token || !activeJiraProject || !epicNameFieldKey) throw new Error('Not configured')
      return createIssue(jiraBaseUrl, token, activeJiraProject, epicName, {
        issuetype: 'Epic',
        description: description || undefined,
        [epicNameFieldKey]: epicName,   // CRITICAL: pass both summary and epicNameFieldKey
      })
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['jira-epics'] })
      setEpicName('')
      setDescription('')
      onClose()
    },
  })

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    if (!epicName.trim()) return
    mutation.mutate()
  }

  return (
    <Dialog.Root open={open} onOpenChange={(o) => { if (!o) onClose() }}>
      <Dialog.Backdrop className="fixed inset-0 bg-black/50 z-40" />
      <Dialog.Popup className="fixed inset-0 z-50 flex items-center justify-center p-4">
        <div className="bg-background rounded-lg shadow-lg w-full max-w-md p-6">
          <Dialog.Title className="text-lg font-semibold mb-4">Create Epic</Dialog.Title>
          <form onSubmit={handleSubmit} className="flex flex-col gap-4">
            <label className="flex flex-col gap-1">
              <span className="text-sm font-medium">Epic Name *</span>
              <input
                aria-label="Epic Name"
                value={epicName}
                onChange={e => setEpicName(e.target.value)}
                className="border rounded px-3 py-2 text-sm"
                required
              />
            </label>
            <label className="flex flex-col gap-1">
              <span className="text-sm font-medium">Description</span>
              <textarea
                value={description}
                onChange={e => setDescription(e.target.value)}
                className="border rounded px-3 py-2 text-sm resize-y min-h-[80px]"
                placeholder="Optional description"
              />
            </label>
            <div className="flex justify-end gap-2 mt-2">
              <Dialog.Close render={<button type="button" className="...">Cancel</button>} />
              <button
                type="submit"
                disabled={!epicName.trim() || mutation.isPending}
                className="..."
                aria-label="Create Epic"
              >
                {mutation.isPending ? 'Creating...' : 'Create Epic'}
              </button>
            </div>
          </form>
        </div>
      </Dialog.Popup>
    </Dialog.Root>
  )
}
```
