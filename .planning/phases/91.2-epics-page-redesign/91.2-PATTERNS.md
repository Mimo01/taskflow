**Per-status-category aggregation + `donePoints` (D-13, D-11)** (jira.ts lines 2483-2493, extend the existing loop — this is a straight port of RESEARCH.md's Code Examples section, already verified against this exact function):
```ts
// CURRENT countMap shape:
const countMap = new Map<string, { total: number; done: number; points: number }>();
for (const story of stories) {
  const ek = story.fields[epicLinkFieldKey] as string | null;
  if (!ek) continue;
  const entry = countMap.get(ek) ?? { total: 0, done: 0, points: 0 };
  entry.total++;
  if (story.fields.status.statusCategory?.key === 'done') entry.done++;
  entry.points += (story.fields[storyPointsFieldKey] as number | null) ?? 0;
  countMap.set(ek, entry);
}

// REQUIRED shape — add inProgress/todo/donePoints:
const countMap = new Map<string, { total: number; done: number; inProgress: number; todo: number; points: number; donePoints: number }>();
for (const story of stories) {
  const ek = story.fields[epicLinkFieldKey] as string | null;
  if (!ek) continue;
  const entry = countMap.get(ek) ?? { total: 0, done: 0, inProgress: 0, todo: 0, points: 0, donePoints: 0 };
  entry.total++;
  const catKey = story.fields.status.statusCategory?.key;
  const sp = (story.fields[storyPointsFieldKey] as number | null) ?? 0;
  entry.points += sp;
  if (catKey === 'done') { entry.done++; entry.donePoints += sp; }
  else if (catKey === 'indeterminate') entry.inProgress++;
  else entry.todo++;
  countMap.set(ek, entry);
}
```
