```typescript
const labelCoverage = useMemo(() => {
  const mrs = milestoneMRs ?? [];
  if (mrs.length === 0) return null;
  const unlabeled = mrs.filter((mr) => mr.labels.length === 0);
  return {
    total: mrs.length,
    labeled: mrs.length - unlabeled.length,
    unlabeled,
    allLabeled: unlabeled.length === 0,
  };
}, [milestoneMRs]);
```
