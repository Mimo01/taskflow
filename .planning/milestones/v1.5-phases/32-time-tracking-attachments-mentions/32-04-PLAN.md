    ```typescript
    function handleTextChange(e: React.ChangeEvent<HTMLTextAreaElement>) {
      const newText = e.target.value;
      setText(newText);

      const textarea = e.target;
      const cursorPos = textarea.selectionStart;

      // Check if @ was just typed or if we're in an active mention
      if (mentionActive) {
        // Extract query text between @ and cursor
        const queryText = newText.slice(mentionStart + 1, cursorPos);
        if (queryText.includes(' ') || cursorPos <= mentionStart) {
          // Space typed or cursor moved before @ — dismiss
          setMentionActive(false);
        } else {
          setMentionQuery(queryText);
        }
      } else {
        // Check if @ was just typed
        const charBefore = cursorPos > 0 ? newText[cursorPos - 1] : '';
        const charBeforeAt = cursorPos > 1 ? newText[cursorPos - 2] : ' '; // treat start of text as space
        if (charBefore === '@' && (charBeforeAt === ' ' || charBeforeAt === '\n' || cursorPos === 1)) {
          setMentionActive(true);
          setMentionStart(cursorPos - 1);
          setMentionQuery('');
          // Calculate cursor position for popover
          const pos = getCursorPixelPosition(textarea, cursorPos - 1);
          setMentionPosition(pos);
        }
      }
    }
    ```

    Cursor position measurement (per RESEARCH.md Textarea Cursor Position pattern):
    ```typescript
    function getCursorPixelPosition(textarea: HTMLTextAreaElement, cursorIndex: number): { top: number; left: number } {
      const mirror = document.createElement('div');
      const style = getComputedStyle(textarea);

      mirror.style.position = 'absolute';
      mirror.style.visibility = 'hidden';
      mirror.style.whiteSpace = 'pre-wrap';
      mirror.style.wordWrap = 'break-word';
      mirror.style.width = style.width;
      mirror.style.font = style.font;
      mirror.style.padding = style.padding;
      mirror.style.border = style.border;
      mirror.style.lineHeight = style.lineHeight;

      const textBefore = textarea.value.substring(0, cursorIndex);
      mirror.textContent = textBefore;
      const marker = document.createElement('span');
      marker.textContent = '|';
      mirror.appendChild(marker);

      document.body.appendChild(mirror);
      const textareaRect = textarea.getBoundingClientRect();
      const mirrorRect = mirror.getBoundingClientRect();
      const markerRect = marker.getBoundingClientRect();

      const position = {
        top: markerRect.top - mirrorRect.top + textarea.offsetHeight + 4,  // Below textarea + gap
        left: markerRect.left - mirrorRect.left,
      };
      document.body.removeChild(mirror);
      return position;
    }
    ```

    Handle mention selection (per D-13):
    ```typescript
    function handleMentionSelect(user: JiraAssignableUser) {
      const textarea = textareaRef.current;
      if (!textarea) return;

      const before = text.slice(0, mentionStart);
      const after = text.slice(textarea.selectionStart);
      const mention = `[~${user.name}]`;
      const newText = `${before}${mention}${after}`;
      const cursorPos = mentionStart + mention.length;

      setText(newText);
      setMentionActive(false);
      setMentionQuery('');
      setMentionStart(-1);

      // Restore cursor position after React re-render (per Pitfall 4 from RESEARCH.md)
      requestAnimationFrame(() => {
        textarea.focus();
        textarea.setSelectionRange(cursorPos, cursorPos);
      });
    }
    ```

      if (mentionActive) {
