Release matching computation via useMemo:
```typescript
// Build candidate list from milestones + tags combined
const candidates = [
  ...milestones.map(m => ({ date: m.due_date, name: m.title, url: m.web_url })),
  ...tags.map(t => ({ date: t.commit.created_at, name: t.name, url: '' })),
];

// For each fix version, find best match (prefer exact over fuzzy)
for (const version of fixVersions) {
  let bestMatch: ReleaseMatch & { url: string } | null = null;
  for (const cand of candidates) {
    const match = matchGitLabToFixVersion(version.releaseDate, cand);
    if (match.type === 'exact') { bestMatch = { ...match }; break; }
    if (match.type === 'fuzzy' && !bestMatch) { bestMatch = { ...match }; }
  }
  // render row with bestMatch
}
```
