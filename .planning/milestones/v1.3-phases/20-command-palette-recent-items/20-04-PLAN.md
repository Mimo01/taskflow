**Full updated component** (no SearchOverlay, no local searchOpen state):
```tsx
import { Bell, Search } from 'lucide-react';
import { Popover, PopoverTrigger, PopoverContent } from '../ui/popover';
import { useUnreadCount } from '../../stores/notifications.store';
import NotificationPopover from '../../routes/notifications/NotificationPopover';
import RecentItemsPopover from './RecentItemsPopover';

interface TopBarProps {
  onIssueClick?: (issueKey: string) => void;
  paletteOpen: boolean;
  onPaletteOpen: () => void;
  notifPopoverOpen: boolean;
  onNotifPopoverChange: (open: boolean) => void;
}

export default function TopBar({ onIssueClick, onPaletteOpen, notifPopoverOpen, onNotifPopoverChange }: TopBarProps) {
  const unreadCount = useUnreadCount();
  return (
    <header className="h-12 border-b flex items-center justify-end px-4 flex-shrink-0 gap-2">
      <button type="button" onClick={onPaletteOpen} aria-label="Search" className="relative flex items-center justify-center w-8 h-8 rounded hover:bg-muted transition-colors">
        <Search className="w-5 h-5" />
      </button>
      <RecentItemsPopover onIssueClick={onIssueClick} />
      <Popover open={notifPopoverOpen} onOpenChange={onNotifPopoverChange}>
        <PopoverTrigger className="relative flex items-center justify-center w-8 h-8 rounded hover:bg-muted transition-colors" aria-label="Notifications">
          <Bell className="w-5 h-5" />
          {unreadCount > 0 && (
            <span className="absolute -top-1 -right-1 bg-red-500 text-white text-xs rounded-full min-w-[1.1rem] h-[1.1rem] flex items-center justify-center px-0.5">
              {unreadCount > 99 ? '99+' : unreadCount}
            </span>
          )}
        </PopoverTrigger>
        <PopoverContent className="p-0 w-80">
          <NotificationPopover onIssueClick={onIssueClick} />
        </PopoverContent>
      </Popover>
    </header>
  );
}
```

Note: The `paletteOpen` prop is accepted but not directly used in TopBar (palette renders in AppLayout). It's passed for future use if TopBar needs to visually indicate palette state. If it causes a lint warning, remove it from the destructured props.
