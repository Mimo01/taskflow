      const { density, setDensity } = useSettingsStore();

      useEffect(() => {
        applyDensity(density);

        <div className="flex flex-col gap-8">
          <h2 className="text-lg font-semibold">Appearance</h2>
          <ThemeSection />
          <div className="flex flex-col gap-3">
