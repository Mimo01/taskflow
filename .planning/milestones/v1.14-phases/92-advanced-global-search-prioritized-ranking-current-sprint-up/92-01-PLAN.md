- `export type HighlightRange = [start: number, end: number]` (half-open, char offsets into the string that is rendered)
- `export interface TextMatch { kind: 'exact' | 'word-start' | 'fuzzy'; score: number; indexes: number[] }` — score in 0..1 (higher better)

- `export function matchText(query: string, target: string, opts?: { fuzzy?: boolean }): TextMatch | null`
- `export function compareTextMatch(a: TextMatch, b: TextMatch): number` — negative when a ranks higher; kind order exact < word-start < fuzzy, then score desc
- `export function runsFromIndexes(indexes: number[], max?: number): HighlightRange[]` — default max 6
- `export function buildSnippet(text: string, query: string, radius?: number): { text: string; ranges: HighlightRange[] } | null` — default radius 48
- `export function stripWikiMarkup(text: string): string`
