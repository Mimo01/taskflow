```typescript
async function handleMoveToSprint() {
  if (!activeSprint || selectedKeys.size === 0) return;
  const keysToMove = Array.from(selectedKeys);
  const previousIssues = queryClient.getQueryData<JiraIssue[]>(['jira-backlog', activeJiraProject, jiraBaseUrl]);
  queryClient.setQueryData<JiraIssue[]>(
    ['jira-backlog', activeJiraProject, jiraBaseUrl],
    (old) => (old ?? []).filter((i) => !selectedKeys.has(i.key))
  );
  setSelectedKeys(new Set());
  setBulkError(null);
  try {
    await addIssuesToSprint(jiraBaseUrl!, jiraToken!, activeSprint.id, keysToMove);
    queryClient.invalidateQueries({ queryKey: ['jira-issues', 'sprint-board'] }); // SprintBoardTab key
    queryClient.invalidateQueries({ queryKey: ['jira-backlog'] });
  } catch {
    queryClient.setQueryData(['jira-backlog', activeJiraProject, jiraBaseUrl], previousIssues);
    setSelectedKeys(new Set(keysToMove));
    setBulkError('Failed to move issues — changes reversed');
  }
}
```

**NOTE on sprint-board invalidation key:** RESEARCH.md open question #2 flags that the sprint board uses key `['jira-issues', 'sprint-board', ...]` NOT `['jira-sprint-issues']`. Use `['jira-issues', 'sprint-board']` as the partial key to invalidate. Verify the actual key in SprintBoardTab.tsx before writing the code.

**NOTE on CreateEditIssueModal onSuccess:** The modal fires query invalidation internally on success. For the backlog to refresh after create, also invalidate `['jira-backlog']` in a `onSuccess` callback passed to the modal OR rely on AppLayout to invalidate after modal close. Simplest: pass `onSuccess` prop through the AppLayout `CreateEditIssueModal` that invalidates backlog cache when the modal was opened via `openCreateStory`. This is an AppLayout change — see Task 1.
