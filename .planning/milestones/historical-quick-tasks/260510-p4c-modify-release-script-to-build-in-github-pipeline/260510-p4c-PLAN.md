**5. Add a `create-release` job that runs after `build` and creates the GitHub release:**

```yaml
create-release:
  needs: build
  if: always() && !cancelled()
  runs-on: ubuntu-latest
  permissions:
    contents: read
  outputs:
    release_id: ${{ steps.create.outputs.release_id }}

  steps:
    - name: Checkout source
      uses: actions/checkout@v4
      with:
        fetch-tags: true
        fetch-depth: 0

    - name: Resolve version
      id: version
      shell: bash
      run: |
        if [ -n "${{ github.event.inputs.version }}" ]; then
          TAG="${{ github.event.inputs.version }}"
        else
          TAG="${{ github.ref_name }}"
        fi
        BARE=$(echo "$TAG" | sed 's/^v//')
        echo "tag=$TAG" >> "$GITHUB_OUTPUT"
        echo "bare=$BARE" >> "$GITHUB_OUTPUT"

    - name: Create GitHub release on taskflow-releases
      id: create
      env:
        RELEASES_TOKEN: ${{ secrets.RELEASES_REPO_PAT }}
      shell: bash
      run: |
        TAG="${{ steps.version.outputs.tag }}"
        VERSION="${{ steps.version.outputs.bare }}"

        # Extract tag annotation body for release notes
        TAG_BODY=$(git tag -l --format='%(contents:body)' "$TAG")
        TAG_BODY_JSON=$(printf '%s' "$TAG_BODY" | python3 -c "import sys,json; print(json.dumps(sys.stdin.read()))")

        RESPONSE=$(curl -s -X POST \
          -H "Authorization: token $RELEASES_TOKEN" \
          -H "Content-Type: application/json" \
          "https://api.github.com/repos/Mimo01/taskflow-releases/releases" \
          -d "{
            \"tag_name\": \"$TAG\",
            \"target_commitish\": \"main\",
            \"name\": \"Taskflow $TAG\",
            \"body\": $TAG_BODY_JSON,
            \"draft\": false,
            \"prerelease\": false
          }")

        RELEASE_ID=$(python3 -c "
        import sys, json
        data = json.loads(sys.argv[1], strict=False)
        if 'id' not in data:
            print('Error creating release. Response:', file=sys.stderr)
            print(json.dumps(data, indent=2), file=sys.stderr)
            sys.exit(1)
        print(data['id'])
        " "$RESPONSE")

        echo "release_id=$RELEASE_ID" >> "$GITHUB_OUTPUT"
        echo "Created release ID: $RELEASE_ID"
        echo "URL: https://github.com/Mimo01/taskflow-releases/releases/tag/$TAG"
```

**7. Replace the "Update latest.json" step with a unified version that builds latest.json from scratch covering all three platforms (macOS, Linux, Windows):**

The current step merges with an existing latest.json (uploaded by local script for macOS). Now CI owns all platforms, so build from scratch:

```yaml
- name: Generate and upload latest.json
  env:
    RELEASES_TOKEN: ${{ secrets.RELEASES_REPO_PAT }}
  shell: bash
  run: |
    VERSION="${{ steps.version.outputs.bare }}"
    TAG="${{ steps.version.outputs.tag }}"
    RELEASE_ID="${{ steps.release.outputs.id }}"
    API_BASE="https://api.github.com/repos/Mimo01/taskflow-releases/releases"
    PUB_DATE=$(date -u +"%Y-%m-%dT%H:%M:%SZ")

    # Extract release notes from git tag
    git fetch --tags 2>/dev/null || true
    TAG_BODY=$(git tag -l --format='%(contents:body)' "$TAG" 2>/dev/null || echo "")

    cat > /tmp/build-latest.py << 'PYEOF'
import json, os, sys
version = sys.argv[1]
tag = sys.argv[2]
pub_date = sys.argv[3]
notes_file = sys.argv[4]

with open(notes_file) as f:
    notes = f.read().strip()

base_url = f"https://github.com/Mimo01/taskflow-releases/releases/download/{tag}"
platforms = {}

# macOS universal
macos_sig_path = f"all-artifacts/macos-artifacts/Taskflow.app.tar.gz.sig"
if os.path.exists(macos_sig_path):
    with open(macos_sig_path) as f:
        macos_sig = f.read().strip()
    macos_entry = {"signature": macos_sig, "url": f"{base_url}/Taskflow.app.tar.gz"}
    platforms["darwin-universal"] = macos_entry
    platforms["darwin-x86_64"] = macos_entry
    platforms["darwin-aarch64"] = macos_entry

# Linux x86_64
# Tauri produces taskflow_VERSION_amd64.AppImage.tar.gz.sig for the updater
import glob
linux_sigs = glob.glob("all-artifacts/linux-artifacts/*.AppImage.tar.gz.sig")
if linux_sigs:
    with open(linux_sigs[0]) as f:
        linux_sig = f.read().strip()
    linux_tgz_name = os.path.basename(linux_sigs[0]).replace(".sig", "")
    platforms["linux-x86_64"] = {"signature": linux_sig, "url": f"{base_url}/{linux_tgz_name}"}

# Windows x86_64
win_sigs = glob.glob("all-artifacts/windows-artifacts/*nsis.zip.sig")
if win_sigs:
    with open(win_sigs[0]) as f:
        win_sig = f.read().strip()
    win_zip_name = os.path.basename(win_sigs[0]).replace(".sig", "")
    platforms["windows-x86_64"] = {"signature": win_sig, "url": f"{base_url}/{win_zip_name}"}

result = {"version": version, "notes": notes, "pub_date": pub_date, "platforms": platforms}
print(json.dumps(result, indent=2))
PYEOF

    printf '%s' "$TAG_BODY" > /tmp/tag-body.txt
    python3 /tmp/build-latest.py "$VERSION" "$TAG" "$PUB_DATE" "/tmp/tag-body.txt" > /tmp/latest.json
    cat /tmp/latest.json

    # Delete old latest.json if present
    ASSETS=$(curl -s -H "Authorization: token $RELEASES_TOKEN" "$API_BASE/$RELEASE_ID/assets")
    LATEST_ID=$(python3 -c "
import sys, json
found = [a['id'] for a in json.loads(sys.argv[1], strict=False) if a['name'] == 'latest.json']
print(found[0] if found else '')
" "$ASSETS")
    if [ -n "$LATEST_ID" ]; then
      echo "Deleting old latest.json (asset $LATEST_ID)..."
      curl -s -X DELETE -H "Authorization: token $RELEASES_TOKEN" "$API_BASE/assets/$LATEST_ID"
    fi

    echo "Uploading latest.json..."
    curl -s -X POST \
      -H "Authorization: token $RELEASES_TOKEN" \
      -H "Content-Type: application/json" \
      "https://uploads.github.com/repos/Mimo01/taskflow-releases/releases/$RELEASE_ID/assets?name=latest.json" \
      --data-binary @/tmp/latest.json | python3 -c "
import sys, json
data = json.loads(sys.stdin.read(), strict=False)
print('OK:', data.get('browser_download_url', '(no url)'))
"
```
